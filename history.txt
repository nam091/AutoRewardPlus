  ✅ Quy trình hoàn thành Mission (dòng 598-662)

  private async completeMission(page: Page, mission: MissionInfo): Promise<void>
  {
      // 1. Click vào mission card để mở trang chi tiết
      await page.evaluate((href) => {
          const links = document.querySelectorAll('a[target="_blank"]')
          for (const link of links) {
              if (link.href === href) {
                  link.click()
                  return true
              }
          }
      }, mission.href)

      // 2. Chờ tab mới mở
      await this.bot.utils.wait(3000)
      const missionPage = await this.bot.browser.utils.getLatestTab(page)

      // 3. Hoàn thành các sub-tasks
      await this.completeMissionTasks(missionPage, mission.title)

      // 4. Đóng tab và quay lại trang /earn
      await this.closeAllExtraTabs(page)
  }

  🎯 Hoàn thành Sub-tasks (dòng 664-719)

  private async completeMissionTasks(missionPage: Page, missionTitle: string):
  Promise<void> {
      const maxAttempts = 3

      for (let attempt = 0; attempt < maxAttempts; attempt++) {
          // Tìm sub-tasks chưa hoàn thành
          const tasks = await this.findMissionSubTasks(missionPage)

          if (!tasks.length) {
              break // Tất cả đã hoàn thành
          }

          // Click từng sub-task
          for (const task of tasks) {
              await this.clickMissionSubTask(missionPage, task)
          }

          // Reload trang để kiểm tra tiến độ
          await missionPage.reload()
          await this.bot.utils.wait(3000)
      }
  }

● ⭐ Chức năng STAR Search - Tổng quan

  STAR = Supplementary Topic-Aware Random Search

  Đây là tính năng tìm kiếm bổ sung chạy SAU khi hoàn thành các search kiếm điểm
  chính, tạo pattern tìm kiếm tự nhiên hơn.

  ---
  🔄 Luồng hoạt động

  1. Tạo N topics đa dạng (ví dụ: 5 topics)
     ├─ Lấy từ nhiều sources: Google, Wikipedia, Reddit, Local
     ├─ Shuffle và chọn top N
     └─ Ví dụ: "best pizza recipes", "climate change solutions", "AI ethics"...

  2. Mở rộng mỗi topic thành sub-queries
     ├─ Lấy Bing suggestions (autocomplete)
     ├─ Lấy related terms
     ├─ Random 3-7 sub-queries mỗi topic
     └─ Ví dụ "pizza recipes" → ["homemade pizza dough", "italian pizza styles",
  ...]

  3. Thực hiện searches
     ├─ Search base topic trước
     ├─ Search các sub-queries
     ├─ Random scroll (50% chance)
     ├─ Random click kết quả (30% chance)
     └─ Delay ngẫu nhiên giữa các searches

  4. Hoàn thành
     └─ Log tổng số searches đã thực hiện

  ---
  🎯 Mục đích

  1. Tự nhiên hóa pattern: Tạo lịch sử tìm kiếm giống người thật
  2. Bổ sung diversity: Không chỉ search các query kiếm điểm
  3. Mô phỏng hành vi: Scroll + click random results
  4. Chạy sau khi kiếm điểm: Không ảnh hưởng đến point earning
